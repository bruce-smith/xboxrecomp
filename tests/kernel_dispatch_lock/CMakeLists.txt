cmake_minimum_required(VERSION 3.20)
project(kernel_dispatch_lock_test C)
if(NOT WIN32)
    message(FATAL_ERROR "The dispatch lock is a Win32 mutex; this test is Windows-only")
endif()
set(CMAKE_C_STANDARD 11)
add_subdirectory(../.. runtime)
add_executable(kernel_dispatch_lock_test test_main.c)
target_include_directories(kernel_dispatch_lock_test PRIVATE ../../src/kernel)
target_link_libraries(kernel_dispatch_lock_test PRIVATE xboxrecomp)
enable_testing()
add_test(NAME kernel_dispatch_lock_test COMMAND kernel_dispatch_lock_test)
# Negative control: with the lock off, the same threads must be seen to overlap.
add_test(NAME kernel_dispatch_lock_off COMMAND kernel_dispatch_lock_test --expect-overlap)
set_tests_properties(kernel_dispatch_lock_off PROPERTIES ENVIRONMENT "RECOMP_DISPATCH_LOCK=0")
